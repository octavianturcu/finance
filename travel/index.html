🗄️ Database setup

Uses your existing people (with relation & photo) and places tables. Run this once to add columns for travel mode & distance.

Show SQL 
-- New columns on the existing places table:
alter table places add column if not exists travel_mode text default 'plane';
alter table places add column if not exists distance_km double precision;

-- Make sure People has relation + photo (skip if already there):
alter table people add column if not exists relation text;
alter table people add column if not exists photo text;

-- (Only if a table is missing) create them:
create table if not exists people (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  relation text,
  photo text,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);
create table if not exists places (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  lat double precision not null,
  lng double precision not null,
  type text default 'travel',
  travel_mode text default 'plane',
  distance_km double precision,
  start_date date,
  end_date date,
  people jsonb default '[]'::jsonb,
  notes text,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);
alter table people enable row level security;
alter table places enable row level security;
drop policy if exists "people_auth" on people;
drop policy if exists "places_auth" on places;
create policy "people_auth" on people for all to authenticated using (true) with check (true);
create policy "places_auth" on places for all to authenticated using (true) with check (true);
