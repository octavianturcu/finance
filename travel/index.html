-- People columns (relation & photo come from your table):
alter table people add column if not exists relation text;
alter table people add column if not exists photo text;
-- If places.id is bigint (invalid input syntax for type bigint), make it text:
alter table places alter column id drop identity if exists;
alter table places alter column id drop default;
alter table places alter column id type text using id::text;
-- Link rows to your account:
alter table places add column if not exists user_id uuid;
alter table people add column if not exists user_id uuid;
-- RLS policies (needed for logged-in read/write):
alter table people enable row level security;
alter table places enable row level security;
drop policy if exists "people_auth" on people;
drop policy if exists "places_auth" on places;
create policy "people_auth" on people for all to authenticated using (true) with check (true);
create policy "places_auth" on places for all to authenticated using (true) with check (true);
